
# Неправильний пароль
$ curl -s -w '\nHTTP %{http_code}\n' -X POST http://localhost:8086/api/auth/login -H 'Content-Type: application/json' -d '{"login":"olena_k","password":"wrong-password"}'
{"error":"invalid_credentials"}
HTTP 401

# Невідомий користувач: та сама відповідь
$ curl -s -w '\nHTTP %{http_code}\n' -X POST http://localhost:8086/api/auth/login -H 'Content-Type: application/json' -d '{"login":"nobody","password":"wrong-password"}'
{"error":"invalid_credentials"}
HTTP 401

# Правильний логін і пароль: токен у відповіді (скорочено)
$ curl -s -w '\nHTTP %{http_code}\n' -X POST http://localhost:8086/api/auth/login -H 'Content-Type: application/json' -d '{"login":"olena_k","password":"Secret2026"}'
{"expiresIn":120,"accessToken":"eyJhbGciOiJIUzI...tEy8yE","tokenType":"Bearer"}
HTTP 200

# Вміст токена (base64url, не шифрування)
$ jwt_decode "$TOKEN"
header:   {"alg": "HS256"}
payload:  {"iss": "car-rental-lr6", "sub": "olena_k", "role": "CLIENT", "iat": 1795296528, "exp": 1795296648, "jti": "218178b2-d91e-46c2-b9cd-b4b6cb346fc8"}
exp - iat = 120 с

# Захищений ендпоінт з токеном

{"login":"olena_k","role":"CLIENT","issuedAt":"2026-11-21T21:28:48Z","expiresAt":"2026-11-21T21:30:48Z","secondsLeft":120}
